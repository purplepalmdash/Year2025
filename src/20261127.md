```
sudo apt install -y nasm
pip3 install --upgrade cmake -i https://pypi.tuna.tsinghua.edu.cn/simple
ninfer-all-master $ ls third_party/media-sources 
dav1d-1.5.4.tar.xz  ffmpeg-9.0.2.tar.xz


sudo apt install -y nasm
sudo apt install -y build-essential libssl-dev zlib1g-dev pkg-config wget
mkdir tmp
cd tmp
wget https://curl.se/download/curl-8.23.0.tar.gz
tar xzvf curl-8.23.0.tar.gz
cd curl-8.23.0

./configure --prefix=/opt/curl --with-openssl --with-ca-bundle=/etc/ssl/certs/ca-certificates.crt\n
make -j12
sudo make install
sudo ldconfig
/opt/curl/bin/curl --version 
PKG_CONFIG_PATH=/opt/curl/lib/pkgconfig pkg-config --modversion libcurl
export PKG_CONFIG_PATH="/opt/curl/lib/pkgconfig:${PKG_CONFIG_PATH:-}"
export CMAKE_PREFIX_PATH="/opt/curl:${CMAKE_PREFIX_PATH:-}"
cd ..
cmake -S . -B build -G Ninja -DCMAKE_BUILD_TYPE=Release -DCMAKE_CUDA_COMPILER=/usr/local/cuda-12.8/bin/nvcc  -DCMAKE_CUDA_ARCHITECTURES=75
cmake --build build --target ninfer-serve ninfer-calibrate
```
