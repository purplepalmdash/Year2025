### 1. non-priv docker instance
test priv script:       

```
#!/bin/bash

# 1. 确保连接并提升为 adb root
echo ">>> 1. 连接容器并获取 Android 内核 Root 权限..."
adb connect 127.0.0.1:5559
adb connect 127.0.0.1:5560

# 提升 adb 权限
adb -s 127.0.0.1:5559 root
adb -s 127.0.0.1:5558 root

# 稍作等待以便 adbd 重启并重新连接
sleep 2
adb connect 127.0.0.1:5559
adb connect 127.0.0.1:5558

echo ""
echo "=================================================="
echo "                  对比测试开始                     "
echo "=================================================="

for PORT in 5559 5558; do
  if [ "$PORT" == "5559" ]; then
    CONTAINER_TYPE="特权容器 (--privileged)"
  else
    CONTAINER_TYPE="非特权容器 (细粒度 --cap-add)"
  fi

  echo -e "\n----------------------------------------"
  echo "【测试目标】: 127.0.0.1:$PORT -> $CONTAINER_TYPE"
  echo "----------------------------------------"

  # 1. Capabilities 测试
  echo -n "[1. CapEff 进程能力位]: "
  adb -s 127.0.0.1:$PORT shell "cat /proc/1/status | grep CapEff" | tr -d '\r'

  # 2. LXCFS 内存测试
  echo -n "[2. MemTotal 挂载内存]: "
  adb -s 127.0.0.1:$PORT shell "cat /proc/meminfo | grep MemTotal" | tr -d '\r'

  # 3. 越权读取宿主磁盘测试 (用 tr -d '\0' 替代 base64，避免误判)
  echo -n "[3. 宿主磁盘 /dev/vda 访问]: "
  RES_DISK=$(adb -s 127.0.0.1:$PORT shell "head -c 10 /dev/vda 2>&1" | tr -d '\0' | tr -d '\r')
  if echo "$RES_DISK" | grep -Eq "Permission denied|No such file|Operation not permitted"; then
    echo "FAILED/BLOCKED ($RES_DISK)"
  else
    echo "SUCCESS (成功读取到磁盘二进制数据)"
  fi

  # 4. 创建设备节点测试 (mknod)
  echo -n "[4. 设备节点创建 (mknod)]: "
  RES_MKNOD=$(adb -s 127.0.0.1:$PORT shell "mknod /dev/test_null c 1 3 && ls -l /dev/test_null && rm /dev/test_null" 2>&1 | tr -d '\r')
  echo "$RES_MKNOD"

  # 5. 挂载文件系统测试 (Mount tmpfs)
  echo -n "[5. 文件系统挂载 (mount)]: "
  RES_MNT=$(adb -s 127.0.0.1:$PORT shell "mkdir -p /data/local/tmp/test_mnt && mount -t tmpfs none /data/local/tmp/test_mnt && echo SUCCESS && umount /data/local/tmp/test_mnt" 2>&1 | tr -d '\r')
  echo "$RES_MNT"

done
```

    CHECKCALL(mount("proc", "/proc", "proc", 0, "hidepid=2,gid=" MAKE_STR(AID_READPROC)));
#undef MAKE_STR

    // 2. 动态检测：只有当 /lxcfs 目录下的文件存在时，才进行覆盖挂载
    const char* lxcfs_files[] = {
        "/proc/meminfo",
        "/proc/cpuinfo",
        "/proc/stat",
        "/proc/uptime"
    };

    for (const char* file : lxcfs_files) {
        std::string src = std::string("/lxcfs") + file;
        // F_OK 用于检查文件是否存在
        if (access(src.c_str(), F_OK) == 0) {
            mount(src.c_str(), file, "none", MS_BIND, NULL);
        }
    }

### 4. redroid packaging script
quickly package redroid instance via:   

```
dash@horse:/media/nvme/vnca13preview/vnca13preview/out/target/product/redroid_x86_64$ cat todocker.sh 
sudo docker rmi redroid13procfix
sudo rm -f `pwd`/redroid13procfix.tar
sudo mount system.img system -o ro
sudo mount vendor.img vendor -o ro
sudo tar --xattrs -c vendor -C system --exclude="./vendor" . |sudo docker import -c 'ENTRYPOINT ["/init", "androidboot.hardware=redroid"]' - redroid13procfix
sudo umount system vendor
sudo docker save -o redroid13procfix.tar redroid13procfix
sudo chmod 777 redroid13procfix.tar 
ls `pwd`/redroid13procfix.tar
```
