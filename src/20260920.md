python3 -c '
from huggingface_hub import HfApi
api = HfApi(endpoint="https://hf-mirror.com")
repo_id = "neroued/Qwen3.8-27B-NInfer"
revision = "3526913004b1cf552cb57b88d6a5c6f5e4a89a70"
for f in api.list_repo_files(repo_id=repo_id, revision=revision):
    print(f"https://hf-mirror.com/{repo_id}/resolve/{revision}/{f}")
' > urls.txt
