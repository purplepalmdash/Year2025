# 20260819
### 1. vllm(qwen38)
Steps:      

```
# 1. 更新系统软件包列表
sudo apt update && sudo apt install -y git git-lfs curl

# 2. 安装 uv 环境管理器
curl -LsSf https://astral.sh/uv/install.sh | sh

# 3. 让 uv 环境变量立即生效
source $HOME/.local/bin/env

# 1. 在主目录下创建名为 vllm-env 的虚拟环境
uv venv ~/vllm-env --python 3.12

# 2. 激活虚拟环境
source ~/vllm-env/bin/activate

# 3. 一键安装 vLLM 及下载依赖（huggingface_hub + hf_transfer 加速）
uv pip install -U vllm --torch-backend=auto
uv pip install -U huggingface_hub hf_transfer modelscope

mkdir -p ~/models
modelscope download --model Qwen/Qwen3.8-27B-FP8 --local_dir ~/models/Qwen3.8-27B-FP8

# start vllm server with mtp enabled. 
vllm serve ~/models/Qwen3.8-27B-FP8   --host 0.0.0.0   --port 8000   --tensor-parallel-size 1   --max-model-len 262144   --kv-cache-dtype fp8   --gpu-memory-utilization 0.90   --enable-prefix-caching   --enable-chunked-prefill   --reasoning-parser qwen3   --tool-call-parser qwen3_coder   --enable-auto-tool-choice   --speculative-config '{"method":"mtp","num_speculative_tokens":3}'

```
