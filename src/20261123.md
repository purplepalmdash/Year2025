### 4. ninfer-all(v100 changes)
Error 定位:      

```
grep -rn "ggml_cuda_kernel_can_use_pdl\|ggml_cuda_get_device\|ggml_cuda_error" src/ third_party/ 2>/dev/null | head
# 看看是否有某个库已经定义了它们（T = 已定义，U = 未定义）
for a in build/src/*/*.a; do nm -C "$a" 2>/dev/null | grep -E " T (ggml_cuda_get_device|ggml_cuda_error)" && echo "  <- $a"; done
```
解决：补一个 shim 文件
新建 src/ops/ggml_cuda_shim.cu，并加到 ninfer_ggml_quants 的源文件列表里：     

```
#include <cuda_runtime.h>
#include <cstdio>
#include <cstdlib>

// 签名必须与 common.cuh 中的声明完全一致（同一命名空间，不要放在 extern "C" 里）
int ggml_cuda_get_device() {
    int id = 0;
    cudaError_t e = cudaGetDevice(&id);
    if (e != cudaSuccess) {
        fprintf(stderr, "cudaGetDevice failed: %s\n", cudaGetErrorString(e));
        abort();
    }
    return id;
}

[[noreturn]] void ggml_cuda_error(const char* stmt, const char* func,
                                  const char* file, int line, const char* msg) {
    int id = -1;
    cudaGetDevice(&id);
    fprintf(stderr, "CUDA error: %s\n  device %d, in %s at %s:%d\n  %s\n",
            msg, id, func, file, line, stmt);
    abort();
}
```
找到 ninfer_ggml_quants 是在哪里定义的：    

```
grep -rn "ninfer_ggml_quants" --include=CMakeLists.txt --include=*.cmake . | grep -v "^./build"

add_library(ninfer_ggml_quants ...) 那条语句是多行的，在它的右括号之后（第 52 行 set_target_properties 之前或之后都可以）加一行：    

```
target_sources(ninfer_ggml_quants PRIVATE ${PROJECT_SOURCE_DIR}/src/ops/ggml_cuda_shim.cu)
```

```

  -DCMAKE_CUDA_COMPILER=/usr/local/cuda-12.8/bin/nvcc -DCMAKE_CUDA_ARCHITECTURES=70
$ cmake --build build --target ninfer-serve ninfer-calibrate
