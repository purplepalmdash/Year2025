### 1. remote vnc
Whole step:    

```
sudo apt update
# 安装 XFCE 桌面环境与 Firefox
sudo apt install -y xfce4 xfce4-goodies firefox dbus-x11
# 安装 TigerVNC 服务端
sudo apt install -y tigervnc-standalone-server tigervnc-common
vncpasswd
mkdir -p ~/.vnc
nano ~/.vnc/xstartup
sudo apt update
sudo apt install -y lxde-core
nano ~/.vnc/xstartup

#!/bin/sh
exec startlxde

sudo snap remove firefox

# 1. 创建存放密钥的目录
sudo install -d -m 0755 /etc/apt/keyrings

# 2. 导入 Mozilla PPA 的官方签名密钥
wget -q https://packages.mozilla.org/apt/repo-signing-key.gpg -O- | sudo tee /etc/apt/keyrings/packages.mozilla.org.asc > /dev/null

# 3. 添加 Mozilla APT 仓库
echo "deb [signed-by=/etc/apt/keyrings/packages.mozilla.org.asc] https://packages.mozilla.org/apt mozilla main" | sudo tee /etc/apt/sources.list.d/mozilla.list > /dev/null

# 4. 设置优先从 Mozilla 仓库下载 Firefox
echo '
Package: *
Pin: origin packages.mozilla.org
Pin-Priority: 1000
' | sudo tee /etc/apt/preferences.d/mozilla

# 5. 更新软件源并安装 Firefox
sudo apt update && sudo apt install -y firefox
```

精简版:     

```
# 1. 更新源并安装 LXDE 桌面、dbus-x11、TigerVNC
sudo apt update
sudo apt install -y lxde-core dbus-x11 tigervnc-standalone-server tigervnc-common

# 2. 设置 VNC 访问密码（会提示输入并确认两次）
vncpasswd

# 3. 配置 VNC 启动脚本（直接写入 LXDE 启动命令）
mkdir -p ~/.vnc
cat << 'EOF' > ~/.vnc/xstartup
#!/bin/sh
exec startlxde
EOF
chmod +x ~/.vnc/xstartup

# 4. 添加 Mozilla 官方 PPA 源并安装标准的 .deb 版 Firefox
sudo install -d -m 0755 /etc/apt/keyrings
wget -q https://packages.mozilla.org/apt/repo-signing-key.gpg -O- | sudo tee /etc/apt/keyrings/packages.mozilla.org.asc > /dev/null
echo "deb [signed-by=/etc/apt/keyrings/packages.mozilla.org.asc] https://packages.mozilla.org/apt mozilla main" | sudo tee /etc/apt/sources.list.d/mozilla.list > /dev/null
echo '
Package: *
Pin: origin packages.mozilla.org
Pin-Priority: 1000
' | sudo tee /etc/apt/preferences.d/mozilla

sudo apt update && sudo apt install -y firefox
```
Start vnc desktop:     

```
vncserver :1 -geometry 1440x900 -depth 24
```

